rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {

    // =========================
    // HELPER FUNCTIONS
    // =========================
    function isSignedIn() {
      return request.auth != null;
    }

    function isOwner(userId) {
      return isSignedIn() && request.auth.uid == userId;
    }

    // Defensive admin check: most profiles store the role in userRole, but
    // some legacy/Google-created profiles only have `role`. Also treats
    // admins/{uid} documents as an admin source of truth when present.
    function isAdmin() {
      return isSignedIn() && (
        get(/databases/$(database)/documents/userProfiles/$(request.auth.uid)).data.userRole == 'admin' ||
        get(/databases/$(database)/documents/userProfiles/$(request.auth.uid)).data.role == 'admin' ||
        exists(/databases/$(database)/documents/admins/$(request.auth.uid))
      );
    }

    // =========================
    // USER PROFILES
    // =========================
    match /userProfiles/{userId} {
      allow read: if true;
      allow create: if isSignedIn();
      allow list: if isSignedIn() && (request.auth.uid == userId || resource.data.referredBy == request.auth.uid || isAdmin());
      
      match /{allPaths=**} {
        allow read, write: if isOwner(userId);
      }

      allow update: if isOwner(userId) ||
                      (isAdmin() &&
                       request.resource.data.diff(resource.data).affectedKeys().hasOnly(['status', 'verificationStatus', 'userRole', 'aiPoints'])) ||
                      (request.resource.data.aiPoints == resource.data.aiPoints + 5 && isSignedIn());
      allow delete: if isAdmin();
    }

    // =========================
    // PRODUCT & LIVE ACCESS
    // =========================
    match /products/{productId} {
      allow read, list: if true;
      allow create: if isSignedIn() && request.resource.data.sellerId == request.auth.uid;
      
      // MERCHANT AUTONOMY: Streamlined owner permission allows all inventory metadata updates.
      // Crucial: We only block unauthorized sellerId changes.
      allow update: if (isOwner(resource.data.sellerId) && (!('sellerId' in request.resource.data) || request.resource.data.sellerId == resource.data.sellerId)) || 
                      (isSignedIn() && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['likes', 'viewCount', 'inCartCount', 'salesCount']));
                      
      allow delete: if isOwner(resource.data.sellerId) || isAdmin();

      match /reviews/{reviewId} {
        allow read: if true;
        allow create: if isSignedIn() && request.resource.data.buyerId == request.auth.uid;
      }
      
      match /comments/{commentId} {
        allow read, list: if true;
        allow create: if isSignedIn();
      }
    }

    match /sellerShops/{shopId} {
      allow read: if true;
      allow update: if isOwner(shopId) && !('followers' in request.resource.data);
      allow write: if isOwner(shopId) || isAdmin();
    }

    match /liveSessions/{sessionId} {
      allow read: if true;
      allow write: if isOwner(sessionId) || isAdmin();
      
      match /chat/{messageId} {
        allow read, write: if isSignedIn();
      }
    }

    // =========================
    // ORDERS & FINANCE
    // =========================
    match /orders/{orderId} {
       // resource-based conditions cannot be evaluated on a bare list() of the
       // whole collection — admins list everything; everyone else needs a filter
       // the rules engine can verify per-document (buyer/seller/reseller).
       allow list, read: if isAdmin()
         || (isSignedIn() && resource.data.buyerId == request.auth.uid)
         || (isSignedIn() && resource.data.sellerId == request.auth.uid)
         || (isSignedIn() && resource.data.resellerId == request.auth.uid);
       allow create: if isSignedIn() && request.resource.data.buyerId == request.auth.uid;
       allow update: if isSignedIn();
       allow delete: if isOwner(resource.data.sellerId) || isAdmin();
    }

    // =========================
    // REQUEST MARKETPLACE (spec §24–39)
    // =========================
    // Requests: briefs are semi-public (sellers browse open requests), but
    // media attachment is private and controlled through /api/media/serve —
    // the browser never touches these collections directly.
    match /requests/{requestId} {
      allow read: if true;
      allow create: if isSignedIn() && request.resource.data.customerId == request.auth.uid;
      allow update: if isAdmin() || isOwner(resource.data.customerId);
      allow delete: if isAdmin() || isOwner(resource.data.customerId);
    }

    match /offers/{offerId} {
      // Offers are visible to their seller, the request's customer, and admins.
      // The customer check uses a get() on the parent request.
      allow read: if isSignedIn() && (
        resource.data.sellerId == request.auth.uid ||
        isAdmin() ||
        get(/databases/$(database)/documents/requests/$(resource.data.requestId)).data.customerId == request.auth.uid
      );
      allow create: if isSignedIn() && request.resource.data.sellerId == request.auth.uid;
      allow update, delete: if isAdmin() || isOwner(resource.data.sellerId);
    }

    // Media records are metadata only; file bytes flow through the
    // authorization-checked /api/media/serve route (admin SDK).
    match /media/{mediaId} {
      allow read: if isSignedIn() && (resource.data.userId == request.auth.uid || isAdmin());
      allow create: if isSignedIn() && request.resource.data.userId == request.auth.uid;
      allow update: if isAdmin() || isOwner(resource.data.userId);
      allow delete: if isAdmin();
    }
    
    match /payouts/{payoutId} {
      allow read: if isSignedIn() && (resource.data.sellerId == request.auth.uid || isAdmin());
      allow create: if isSignedIn() && request.resource.data.sellerId == request.auth.uid;
      allow update: if isAdmin();
    }

    match /walletTransactions/{transactionId} {
        allow create: if false; 
        allow read, list: if isSignedIn() && (request.auth.uid == resource.data.userId || isAdmin());
    }

    // =========================
    // SYSTEM & MARKETING
    // =========================
    match /homepageSlides/{slideId} {
        allow read, list: if true;
        allow write: if isAdmin();
    }
    
    match /helpArticles/{articleId} {
        allow read, list: if true;
        allow write: if isAdmin();
    }

    match /supportTickets/{ticketId} {
      allow create: if isSignedIn() && request.resource.data.userId == request.auth.uid;
      allow list, read: if isSignedIn() && (request.auth.uid == resource.data.userId || isAdmin());
      allow update, delete: if isAdmin();
    }

    match /complaints/{complaintId} {
      allow create: if isSignedIn() && request.resource.data.complainantId == request.auth.uid;
      allow read, update, delete: if isAdmin();
    }

    match /chats/{chatId} {
      allow list, read, update: if isSignedIn() && (request.auth.uid in resource.data.participantIds || isAdmin());
      allow create: if isSignedIn() && request.resource.data.participantIds.size() == 2 && request.auth.uid in request.resource.data.participantIds;
      match /messages/{messageId} {
        allow read, write: if isSignedIn();
      }
    }

    match /notifications/{notificationId} {
      allow read, write: if isSignedIn();
    }

    match /payments/{paymentId} {
      allow read, create: if isSignedIn();
    }

    match /errorLogs/{errorId} {
      allow create: if true;
      allow read, list: if isAdmin();
    }

    // =========================
    // ADMIN SYSTEM LOGS & QUEUES
    // (written by Functions/admin SDK; admins read from the dashboard)
    // =========================
    match /activityLogs/{logId} {
      allow read, list: if isAdmin();
      allow create: if isSignedIn();
    }

    match /admins/{adminId} {
      allow read, list: if isAdmin();
    }

    match /contentModerationQueue/{itemId} {
      allow read, list: if isAdmin();
      allow create: if isSignedIn();
    }

    match /fraudAlerts/{alertId} {
      allow read, list: if isAdmin();
    }

    match /adminChat/{chatId} {
      allow read, list: if isAdmin();
      match /messages/{messageId} {
        allow read, write: if isAdmin();
      }
    }

    match /resellActivities/{activityId} {
      allow read, list: if isSignedIn() && (resource.data.resellerId == request.auth.uid || resource.data.sellerId == request.auth.uid || isAdmin());
      allow create: if isSignedIn() && (request.resource.data.resellerId == request.auth.uid || request.resource.data.sellerId == request.auth.uid);
    }

    match /conversations/{conversationId} {
      allow read, list, update: if isSignedIn() && (request.auth.uid in resource.data.participantIds || isAdmin());
      allow create: if isSignedIn();
    }

    // =========================
    // COMMUNICATION ENGINE (Vidnov)
    // =========================
    match /emailLogs/{logId} {
      allow read, list: if isAdmin();
    }

    match /devMailbox/{mailId} {
      allow read, list: if isAdmin();
    }

    match /securityEvents/{eventId} {
      allow read, list: if isAdmin();
      allow create: if isSignedIn();
    }

    match /userSessions/{sessionId} {
      allow create: if isSignedIn() && request.resource.data.uid == request.auth.uid;
      allow read, update, delete: if isSignedIn() && resource.data.uid == request.auth.uid;
    }

    // passwordResetTokens / passwordResetRequests / passwordResetRequests
    // are intentionally NOT readable by any client — the admin SDK only.

    // =========================
    // DEFAULT DENY
    // =========================
    match /{document=**} {
      allow read, write: if false;
    }
  }
}
